    let engine = PaperEngine::new(storage);
    engine.create_account("admin", 10000.0);

        "id": "test_1",
        "user_id": "admin",

        "side": "Buy",
        "position_side": "Long",
        "order_type": "Limit",
        "price": 60000.0,
        "stop_price": 0.0,
        "amount": 0.1,
        "leverage": 20.0,
        "executed": 0.0,
        "timestamp": 0
    });
    let order: Order = serde_json::from_value(order_json).unwrap();
    engine.submit_order("admin", order).unwrap();
