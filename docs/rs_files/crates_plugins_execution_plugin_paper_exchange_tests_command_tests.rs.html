FONKSİYON test_cancel_order_and_cancel_all() -> ():
    let engine = make_engine();

    id: "ord_1".into(),
    user_id: "admin".into(),

    order_type: OrderType::Limit,

    executed: 0.0,
    timestamp: 0,

    user_id: "admin".into(),
    symbol: "ETHUSDT".into(),
    side: OrderSide::Sell,
    position_side: PositionSide::Short,
    order_type: OrderType::Limit,
    price: 3500.0,
    stop_price: 0.0,
    amount: 2.0,
    leverage: 20.0,
    executed: 0.0,
    timestamp: 0,

    engine.submit_order("admin", o2).unwrap();

    assert!(cancelled);
    assert_eq!(engine.active_orders.get("BTCUSDT").unwrap().len(), 0);
    // Cancel all remaining orders
    let count = engine.cancel_all_orders(None);
    assert_eq!(count, 1);
    assert_eq!(engine.active_orders.get("ETHUSDT").unwrap().len(), 0);
